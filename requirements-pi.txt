# Dependencies to install ON the Raspberry Pi 5 (Raspberry Pi OS, 64-bit).
# Install system packages first (libcamera/picamera2 ship as apt packages,
# not reliably via pip):
#
#   sudo apt update
#   sudo apt install -y python3-picamera2 python3-opencv python3-numpy python3-pip
#
# Then, inside a venv created with --system-site-packages (so picamera2 and
# the apt-installed opencv/numpy are visible):
#
#   python3 -m venv --system-site-packages ~/lockon-venv
#   source ~/lockon-venv/bin/activate
#   pip install -r requirements-pi.txt
#
